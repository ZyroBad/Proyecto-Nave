cmake_minimum_required(VERSION 3.16)
project(nave_cb_renderer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SDL2 QUIET CONFIG)
add_executable(nave_renderer main.cpp framebuffer.cpp triangle.cpp obj_loader.cpp)
target_include_directories(nave_renderer PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
if(TARGET SDL2::SDL2)
  target_link_libraries(nave_renderer PRIVATE SDL2::SDL2)
  target_compile_definitions(nave_renderer PRIVATE NAVE_HAS_SDL2=1)
elseif(TARGET SDL2::SDL2-static)
  target_link_libraries(nave_renderer PRIVATE SDL2::SDL2-static)
  target_compile_definitions(nave_renderer PRIVATE NAVE_HAS_SDL2=1)
else()
  message(WARNING "SDL2 no encontrado: solo funcionará el modo --render sin ventana. Instala SDL2 para abrir la ventana.")
  target_compile_definitions(nave_renderer PRIVATE NAVE_HAS_SDL2=0)
endif()
